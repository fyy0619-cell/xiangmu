# mpu6050_iio 驱动 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 MPU-6050 写一个从零自研的 Linux IIO 驱动,跑在 STM32MP157 上,支持 sysfs 单次读取与 DRDY 硬件中断触发的缓冲流式数据。

**Architecture:** 单文件 i2c 驱动 `mpu6050_iio.c`,通过设备树匹配。probe 用 regmap 访问寄存器并校验 WHO_AM_I;实现 IIO 通道的 read_raw/write_raw;用 `iio_triggered_buffer` 挂 kfifo 缓冲,先软件触发跑通,再接 GPIO 中断做 DRDY 硬件触发。

**Tech Stack:** Linux IIO 子系统、regmap-i2c、设备树、devm 资源管理、libiio(userspace 验证)。

## Global Constraints

- 目标板:STM32MP157 PRO(100ASK),内核源码与交叉工具链在 Ubuntu VM(`ssh vm`,sudo 密码 123456)。
- 板↔VM 直连:板 192.168.2.250 / VM 192.168.2.60,TFTP/NFS 加载模块;串口 CH340 = COM10 @ 115200。
- 驱动作为**可加载模块** `.ko` 开发(便于反复 insmod/rmmod),不编进 zImage。
- 内核需开启:`CONFIG_IIO=y`、`CONFIG_IIO_BUFFER=y`、`CONFIG_IIO_TRIGGERED_BUFFER=y`、`CONFIG_IIO_SYSFS_TRIGGER=y`、`CONFIG_REGMAP_I2C=y`。若为模块则一并加载。
- MPU6050 关键寄存器:`WHO_AM_I=0x75`(值 0x68)、`PWR_MGMT_1=0x6B`、`SMPLRT_DIV=0x19`、`CONFIG=0x1A`、`GYRO_CONFIG=0x1B`、`ACCEL_CONFIG=0x1C`、`INT_PIN_CFG=0x37`、`INT_ENABLE=0x38`、`INT_STATUS=0x3A`、数据区 `ACCEL_XOUT_H=0x3B` 起连续 14 字节(accel6 + temp2 + gyro6)。
- 驱动 `compatible = "invensense,mpu6050"`,I2C 地址 `0x68`。
- 代码风格:全程 `devm_` 管理资源;错误路径用 `dev_err_probe()`;不手写 remove 释放。

```c
// drivers/mpu6050_iio/mpu6050_iio.c
#include <linux/module.h>
#include <linux/i2c.h>
#include <linux/mod_devicetable.h>
#include <linux/regmap.h>
#include <linux/mutex.h>

#define MPU6050_REG_WHO_AM_I   0x75
#define MPU6050_REG_PWR_MGMT_1 0x6B
#define MPU6050_WHO_AM_I_VAL   0x68
#define MPU6050_PWR1_SLEEP     BIT(6)

struct mpu6050_data {
	struct regmap *regmap;
	struct mutex lock;
	struct i2c_client *client;
};

static const struct regmap_config mpu6050_regmap_config = {
	.reg_bits = 8,
	.val_bits = 8,
	.max_register = 0x75,
};

static int mpu6050_probe(struct i2c_client *client)
{
	struct device *dev = &client->dev;
	struct mpu6050_data *data;
	unsigned int who;
	int ret;

	data = devm_kzalloc(dev, sizeof(*data), GFP_KERNEL);
	if (!data)
		return -ENOMEM;

	data->client = client;
	mutex_init(&data->lock);
	i2c_set_clientdata(client, data);

	data->regmap = devm_regmap_init_i2c(client, &mpu6050_regmap_config);
	if (IS_ERR(data->regmap))
		return dev_err_probe(dev, PTR_ERR(data->regmap),
				     "regmap init failed\n");

	ret = regmap_read(data->regmap, MPU6050_REG_WHO_AM_I, &who);
	if (ret)
		return dev_err_probe(dev, ret, "WHO_AM_I read failed\n");
	if (who != MPU6050_WHO_AM_I_VAL)
		return dev_err_probe(dev, -ENODEV,
				     "bad WHO_AM_I 0x%02x (want 0x%02x)\n",
				     who, MPU6050_WHO_AM_I_VAL);

	/* 唤醒:清 sleep 位 */
	ret = regmap_update_bits(data->regmap, MPU6050_REG_PWR_MGMT_1,
				 MPU6050_PWR1_SLEEP, 0);
	if (ret)
		return dev_err_probe(dev, ret, "wakeup failed\n");

	dev_info(dev, "mpu6050 probed, WHO_AM_I=0x%02x\n", who);
	return 0;
}

static const struct of_device_id mpu6050_of_match[] = {
	{ .compatible = "invensense,mpu6050" },
	{ }
};
MODULE_DEVICE_TABLE(of, mpu6050_of_match);

static const struct i2c_device_id mpu6050_id[] = {
	{ "mpu6050", 0 },
	{ }
};
MODULE_DEVICE_TABLE(i2c, mpu6050_id);

static struct i2c_driver mpu6050_driver = {
	.driver = {
		.name = "mpu6050_iio",
		.of_match_table = mpu6050_of_match,
	},
	.probe = mpu6050_probe,
	.id_table = mpu6050_id,
};
module_i2c_driver(mpu6050_driver);

MODULE_AUTHOR("fyy0619-cell");
MODULE_DESCRIPTION("Invensense MPU-6050 IIO driver (from scratch)");
MODULE_LICENSE("GPL");
```

> 注:较老内核(<5.3)`probe` 原型是 `probe(client, id)`。若编译报原型不符,改为 `mpu6050_probe(struct i2c_client *client, const struct i2c_device_id *id)`。

- [ ] **Step 3: 实现单寄存器读取辅助 + read_raw(仅 RAW)**

```c
/* 读一个 16-bit 大端有符号量:reg 为高字节地址 */
static int mpu6050_read_reg16(struct mpu6050_data *data, u8 reg, int *val)
{
	__be16 raw;
	int ret;

	ret = regmap_bulk_read(data->regmap, reg, &raw, sizeof(raw));
	if (ret)
		return ret;
	*val = (s16)be16_to_cpu(raw);
	return 0;
}

static int mpu6050_read_raw(struct iio_dev *indio_dev,
			    struct iio_chan_spec const *chan,
			    int *val, int *val2, long mask)
{
	struct mpu6050_data *data = iio_priv(indio_dev);
	int ret, raw;
	u8 reg;

	switch (mask) {
	case IIO_CHAN_INFO_RAW:
		switch (chan->type) {
		case IIO_ACCEL:
			reg = MPU6050_REG_ACCEL_XOUT_H + (chan->channel2 - IIO_MOD_X) * 2;
			break;
		case IIO_ANGL_VEL:
			reg = MPU6050_REG_GYRO_XOUT_H + (chan->channel2 - IIO_MOD_X) * 2;
			break;
		case IIO_TEMP:
			reg = MPU6050_REG_TEMP_OUT_H;
			break;
		default:
			return -EINVAL;
		}
		mutex_lock(&data->lock);
		ret = mpu6050_read_reg16(data, reg, &raw);
		mutex_unlock(&data->lock);
		if (ret)
			return ret;
		*val = raw;
		return IIO_VAL_INT;
	default:
		return -EINVAL;
	}
}

static const struct iio_info mpu6050_info = {
	.read_raw = mpu6050_read_raw,
};
```

> `IIO_MOD_X/Y/Z` 连续,故轴偏移用 `channel2 - IIO_MOD_X`。

- [ ] **Step 4: 实现 write_raw(采样率),映射到 SMPLRT_DIV**

MPU6050 陀螺输出率基准 1kHz(DLPF 开启时),`rate = 1000/(1+SMPLRT_DIV)`。

```c
static int mpu6050_set_samp_freq(struct mpu6050_data *data, int freq)
{
	unsigned int div;

	if (freq <= 0 || freq > 1000)
		return -EINVAL;
	div = 1000 / freq - 1;
	if (div > 255)
		div = 255;
	return regmap_write(data->regmap, MPU6050_REG_SMPLRT_DIV, div);
}

static int mpu6050_write_raw(struct iio_dev *indio_dev,
			     struct iio_chan_spec const *chan,
			     int val, int val2, long mask)
{
	struct mpu6050_data *data = iio_priv(indio_dev);
	int ret;

	switch (mask) {
	case IIO_CHAN_INFO_SAMP_FREQ:
		mutex_lock(&data->lock);
		ret = mpu6050_set_samp_freq(data, val);
		if (!ret)
			data->samp_freq = val;
		mutex_unlock(&data->lock);
		return ret;
	default:
		return -EINVAL;
	}
}
```

在 `mpu6050_info` 中加:`.write_raw = mpu6050_write_raw,`

- [ ] **Step 2: 实现 pollfunc:一次 burst 读 14 字节并推入缓冲**

```c
static irqreturn_t mpu6050_trigger_handler(int irq, void *p)
{
	struct iio_poll_func *pf = p;
	struct iio_dev *indio_dev = pf->indio_dev;
	struct mpu6050_data *data = iio_priv(indio_dev);
	/* 14 字节数据 + 时间戳对齐;放大到能容纳 8 通道 */
	struct {
		__be16 chans[7];   /* ax,ay,az,temp,gx,gy,gz */
		s64 ts __aligned(8);
	} scan;
	int ret;

	mutex_lock(&data->lock);
	ret = regmap_bulk_read(data->regmap, MPU6050_REG_ACCEL_XOUT_H,
			       scan.chans, sizeof(scan.chans));
	mutex_unlock(&data->lock);
	if (ret)
		goto done;

	iio_push_to_buffers_with_timestamp(indio_dev, &scan,
					   iio_get_time_ns(indio_dev));
done:
	iio_trigger_notify_done(indio_dev->trig);
	return IRQ_HANDLED;
}
```

/* test/read_raw.c —— 读 accel 三轴 raw 并按 scale 换算成 m/s^2 */
#include <stdio.h>
#include <stdlib.h>

static double read_d(const char *path) {
	FILE *f = fopen(path, "r");
	double v = 0;
	if (!f) { perror(path); exit(1); }
	if (fscanf(f, "%lf", &v) != 1) { fprintf(stderr, "parse %s\n", path); exit(1); }
	fclose(f);
	return v;
}

int main(int argc, char **argv) {
	const char *base = argc > 1 ? argv[1]
		: "/sys/bus/iio/devices/iio:device0";
	char p[256];
	double scale, x, y, z;
	snprintf(p, sizeof p, "%s/in_accel_scale", base); scale = read_d(p);
	snprintf(p, sizeof p, "%s/in_accel_x_raw", base); x = read_d(p) * scale;
	snprintf(p, sizeof p, "%s/in_accel_y_raw", base); y = read_d(p) * scale;
	snprintf(p, sizeof p, "%s/in_accel_z_raw", base); z = read_d(p) * scale;
	printf("accel [m/s^2]  x=%.3f  y=%.3f  z=%.3f  |mag|=%.3f\n",
	       x, y, z, __builtin_sqrt(x*x + y*y + z*z));
	return 0;
}

/* test/stream_buffer.c —— 直接 read() /dev/iio:deviceN 打印原始扫描帧
 * 用法:先用 sysfs 使能通道+trigger+buffer/enable,再运行本程序。
 */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>

int main(int argc, char **argv) {
	const char *dev = argc > 1 ? argv[1] : "/dev/iio:device0";
	int fd = open(dev, O_RDONLY);
	unsigned char buf[64];
	if (fd < 0) { perror(dev); return 1; }
	for (int n = 0; n < 10; n++) {
		ssize_t r = read(fd, buf, sizeof buf);
		if (r <= 0) { perror("read"); break; }
		printf("frame %2d: %zd bytes:", n, r);
		for (ssize_t i = 0; i < r; i++) printf(" %02x", buf[i]);
		putchar('\n');
	}
	close(fd);
	return 0;
}
